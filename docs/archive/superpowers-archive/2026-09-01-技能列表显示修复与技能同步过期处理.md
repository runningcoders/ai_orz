```rust
    async fn install_to_agent(
        &self,
        ctx: RequestContext,
        source_skill_id: &str,
        agent_id: &str,
    ) -> Result<Skill> {
        let ctx = ctx.to_builder().agent_id(agent_id).build();
        // 1. 获取源技能 PO
        let source_po = self
            .skill_dao
            .find_by_id(ctx.clone(), source_skill_id)
            .await?
            .ok_or_else(|| err!(ResourceNotFound, "Skill not found"))?;
        let source_files = self.skill_dao.list_files(&source_po)?;
        let source = Skill {
            po: source_po,
            files: source_files,
            search_match: None,
        };

        // 2. 查找「任意」现有副本（不排除 Expired）——目的：旧副本存在时原地复用旧 ID 覆盖，
        //    避免 (author_id, parent_skill_id) 出现 Expired + Draft 两行并存堆积。
        let existing_any = self
            .skill_dao
            .query(
                ctx.clone(),
                SkillQuery {
                    author_id: Some(agent_id.to_string()),
                    parent_skill_id: Some(source_skill_id.to_string()),
                    // ⚠️ 注意：此处不过滤状态，与「幂等返回」需要区分
                    // 因为下面会再按状态分两个分支处理。
                    ..Default::default()
                },
            )
            .await?;

        if let Some(existing_po) = existing_any.items.into_iter().next() {
            // =========================================================
            // 分支 A：找到现有副本（不管状态）→ 原地更新
            //   - 内容：覆盖写文件（与 overwrite_skill_copy 策略一致）
            //   - 元数据：name/description/tags 对齐源技能
            //   - 状态：无论旧状态是 Expired/Draft/Published → 统一重置为 Draft
            //     （Agent 私有副本安装成功后的目标状态即为 Draft）
            // =========================================================
            let is_status_change = existing_po.status != SkillStatus::Draft;
            let is_meta_change = existing_po.name != source.po.name
                || existing_po.description != source.po.description
                || existing_po.tags != source.po.tags;

            // A1. 逐个文件做内容对比（直接字节比较，避免引入额外 hash 依赖）
            //     - 内容相同 → 跳过写（省 IO + 不抖动时间戳）
            //     - 内容不同 / 副本文件缺失 → 覆盖写（强制对齐源技能最新内容）
            //     - 过期副本保留文件机制正是这个快速路径的前提：
            //       文件几乎都还在，对比后常可整包跳过写 IO。
            use std::borrow::Cow;
            let mut any_file_changed = false;
            for file in &source.files {
                // A1a. 取得源文件字节（内存 content 直接借；大文件按需读成 owned）
                let src_bytes: Cow<[u8]> = if let Some(content) = &file.content {
                    Cow::Borrowed(content.as_bytes())
                } else {
                    let owned = match self.skill_dao.read_file_bytes(&source.po, &file.filename) {
                        Ok(b) => b,
                        Err(_) => {
                            // bytes 方法不可用时降级到 String 版本（SkillDao trait 必定提供 read_file）
                            let s = self.skill_dao.read_file(&source.po, &file.filename)?;
                            s.into_bytes()
                        }
                    };
                    Cow::Owned(owned)
                };

                // A1b. 读取副本侧文件字节（读取失败视为文件不存在/损坏 → 需要写）
                let dst_bytes_opt = self
                    .skill_dao
                    .read_file_bytes(&existing_po, &file.filename)
                    .ok();
                let dst_matches = dst_bytes_opt
                    .as_ref()
                    .map(|b| b.as_slice() == src_bytes.as_ref())
                    .unwrap_or(false);

                // A1c. 不一致才真写入
                if !dst_matches {
                    any_file_changed = true;
                    let write_ok = self
                        .skill_dao
                        .write_file_bytes(&existing_po, &file.filename, &src_bytes);
                    if write_ok.is_err() {
                        // bytes 写入失败 → 降级 String 写入兜底
                        let s = String::from_utf8_lossy(&src_bytes).to_string();
                        self.skill_dao
                            .write_file(&existing_po, &file.filename, &s)?;
                    }
                }
            }

            // A2. 构造更新 PO（任一类变更触发统一一次 DB UPDATE）
            let needs_update = is_status_change || is_meta_change || any_file_changed;
            let updated_po = if needs_update {
                let mut po = existing_po.clone();
                // 统一打一次 updated_at，避免多字段各自重复赋值
                po.updated_at = common::time::now_secs();
                // 状态复活：Expired/Draft/Published → 统一 Draft（Agent 私有副本）
                if is_status_change {
                    po.status = SkillStatus::Draft;
                }
                if existing_po.name != source.po.name {
                    po.name = source.po.name.clone();
                }
                if existing_po.description != source.po.description {
                    po.description = source.po.description.clone();
                }
                if existing_po.tags != source.po.tags {
                    po.tags = source.po.tags.clone();
                }
                self.skill_dao.update(ctx.clone(), &po).await?;
                po
            } else {
                existing_po.clone()
            };

            log_info!(
                &ctx,
                "install_to_agent",
                "复用现有副本并原地更新: source_skill_id={}, agent_id={}, existing_id={}, status_from={:?}, needs_update={}, status_change={}, meta_change={}, file_change={}",
                source_skill_id,
                agent_id,
                existing_po.id,
                existing_po.status,
                needs_update,
                is_status_change,
                is_meta_change,
```

```rust
// ============================================================================
// 资源级权限辅助：Skill 文件/更新/删除 统一访问控制（5 处复用）
// ============================================================================
impl HrDomainImpl {
    /// Skill 资源访问权限判定（对"文件内容读写 / 元数据更新 / 删除"统一入口）。
    ///
    /// **放行条件（满足任一即 Return Ok）：**
    /// 1. 用户角色 ≥ Admin（SuperAdmin(0) / Admin(1)，并查集 Admin owns Member）
    /// 2. 请求用户就是技能作者（po.author_id == uid）
    /// 3. 技能作者类型是 Agent，且请求用户是该 Agent 的创建者（Agent.created_by == uid）
    ///
    /// 所有其他情形 → 返回 ErrorCode::Forbidden（HTTP 403）+ 解释性错误信息。
    ///
    /// Context 补充原则符合性：Agent 查询仅在"不是管理员+不是作者"时才执行，
    /// 属于"下游明确需要，且当前上下文未包含"的场景下才专门查询。
    async fn ensure_skill_access(&self, ctx: &RequestContext, po: &SkillPo) -> Result<()> {
        use common::enums::UserRole;
        use common::enums::skill::SkillAuthorType;

        let uid = ctx.uid();

        // ---- ① Admin Bypass ----
        if let Some(role) = ctx.user_role() {
            match UserRole::from(role) {
                UserRole::SuperAdmin | UserRole::Admin => return Ok(()),
                UserRole::Member => {} // 走后续检查
            }
        }

        // ---- ② 作者本人 ----
        if !uid.is_empty() && po.author_id == uid {
            return Ok(());
        }

        // ---- ③ Agent 创建者 ----
        if matches!(po.author_type, SkillAuthorType::Agent) && !po.author_id.is_empty() {
            if let Some(agent) = self
                .agent_dal
                .find_by_id(ctx.clone(), &po.author_id)
                .await?
            {
                if agent.po.created_by == uid {
                    return Ok(());
                }
            }
        }

        // ---- 拒绝 ----
        bail_err!(
            Forbidden,
            "无权访问 Skill {id}（作者类型={atype:?}，作者={author}）。\
             可访问者：技能作者 / Agent 创建者（当作者类型为 Agent 时）/ 管理员",
            id = po.id,
            atype = po.author_type,
            author = po.author_id,
        );
    }
}
```
