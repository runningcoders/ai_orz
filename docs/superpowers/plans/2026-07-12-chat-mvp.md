> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

```rust
//! Message API DTOs - 消息列表查询

use ai_orz_macros::Params;
use schemars::JsonSchema;
use serde::{Deserialize, Serialize};

/// 消息列表查询请求（GET query params）
///
/// 支持两种分页模式：
/// 1. 初始加载 / 上拉翻页：`before_timestamp` + `limit` + `order=desc` → 获取更早的消息
/// 2. 下拉轮询新消息：`after_timestamp` → 获取更新的消息
#[derive(Debug, Clone, Default, Deserialize, Serialize, JsonSchema, Params)]
pub struct ListMessagesRequest {
    /// 按项目 ID 过滤
    #[serde(skip_serializing_if = "Option::is_none")]
    pub project_id: Option<String>,
    /// 按任务 ID 过滤
    #[serde(skip_serializing_if = "Option::is_none")]
    pub task_id: Option<String>,
    /// 按发送方 ID 过滤
    #[serde(skip_serializing_if = "Option::is_none")]
    pub from_id: Option<String>,
    /// 按接收方 ID 过滤
    #[serde(skip_serializing_if = "Option::is_none")]
    pub to_id: Option<String>,
    /// 上拉翻页：只返回 created_at 小于此值的消息（毫秒时间戳）
    /// 用于加载更早的历史消息
    #[serde(skip_serializing_if = "Option::is_none")]
    pub before_timestamp: Option<i64>,
    /// 下拉轮询：只返回 created_at 大于此值的消息（毫秒时间戳）
    /// 用于增量拉取新消息
    #[serde(skip_serializing_if = "Option::is_none")]
    pub after_timestamp: Option<i64>,
    /// 限制返回条数（默认 10）
    #[serde(skip_serializing_if = "Option::is_none")]
    pub limit: Option<usize>,
}

/// 消息列表项（脱敏后的展示对象）
#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
pub struct MessageListItem {
    /// 消息 ID
    pub message_id: String,
    /// 关联项目 ID
    #[serde(skip_serializing_if = "Option::is_none")]
    pub project_id: Option<String>,
    /// 关联任务 ID
    #[serde(skip_serializing_if = "Option::is_none")]
    pub task_id: Option<String>,
    /// 发送方 ID
    pub from_id: String,
    /// 发送方角色（0=User, 1=Agent, 2=System）
    pub from_role: i32,
    /// 接收方 ID
    pub to_id: String,
    /// 接收方角色
    pub to_role: i32,
    /// 消息类型（0=Text, 5=ToolCallRequest, 6=ToolCallResult, 9=TaskAssignment 等）
    pub message_type: i32,
    /// 消息状态（0=Recalled, 1=Pending, 2=Processing, 3=Processed, 4=Failed）
    pub status: i32,
    /// 消息内容
    pub content: String,
    /// 回复的消息 ID
    #[serde(skip_serializing_if = "Option::is_none")]
    pub reply_to_id: Option<String>,
    /// 创建时间戳（毫秒）
    pub created_at: i64,
}

/// 消息列表响应
#[derive(Debug, Clone, Default, Serialize, Deserialize, JsonSchema)]
pub struct ListMessagesResponse {
    /// 消息列表（按 created_at ASC 排序）
    pub messages: Vec<MessageListItem>,
    /// 总数（当前页条数）
    pub total: usize,
}
```

```rust
//! Handler: GET /api/v1/messages - List messages with filtering

use crate::pkg::RequestContext;
use crate::service::dao::message::MessageQuery;
use crate::service::domain::message;
use ai_orz_macros::{generate_http_handler, register_handler_tool};
use common::api::{ListMessagesRequest, ListMessagesResponse, MessageListItem};
use common::error::{Result, err, bail_err};

/// List messages with optional filtering by project, task, from_id, to_id, before/after timestamp
///
/// 分页模式：
/// - 初始加载 / 上拉翻页：传 `before_timestamp` → 返回 created_at < before_timestamp 的消息，按 DESC 排序
/// - 下拉轮询新消息：传 `after_timestamp` → 返回 created_at > after_timestamp 的消息，按 ASC 排序
/// - 无时间过滤：默认返回最新消息，按 DESC 排序
#[register_handler_tool(
    id = "list_messages",
    name = "list_messages",
    description = "List messages with optional filtering by project, task, sender, receiver, with bidirectional pagination",
    params = "common::api::ListMessagesRequest"
)]
#[generate_http_handler]
pub async fn list_messages(
    ctx: RequestContext,
    params: ListMessagesRequest,
) -> Result<ListMessagesResponse> {
    let org_id = ctx
        .organization_id
        .clone()
        .ok_or_else(|| err!(InvalidRequest, "当前请求缺少组织上下文"))?;
    let user_id = ctx.uid();
    if user_id.is_empty() {
        bail_err!(InvalidRequest, "当前请求缺少用户上下文");
    }

    let limit = params.limit.unwrap_or(10);

    // 判断排序方向：有 after_timestamp 时 ASC（新消息追加），否则 DESC（历史翻页）
    let order_by = if params.after_timestamp.is_some() {
        "created_at ASC".to_string()
    } else {
        "created_at DESC".to_string()
    };

    // 构建 DAO 层查询条件
    // before_timestamp 和 after_timestamp 在 Handler 层过滤（DAO 层不支持这两个字段）
    let query = MessageQuery {
        organization_id: Some(org_id),
        project_id: params.project_id.clone(),
        task_id: params.task_id.clone(),
        from_id: params.from_id.clone(),
        to_id: params.to_id.clone(),
        limit: Some(limit + 100), // 多拉一些，Handler 层再按时间过滤
        offset: None,
        order_by: Some(order_by),
        ..Default::default()
    };

    let messages = message::domain().management().query(ctx, query).await?;

    // Handler 层时间过滤
    let filtered: Vec<_> = match (params.before_timestamp, params.after_timestamp) {
        (Some(before), None) => {
            // 上拉翻页：created_at < before
            messages
                .into_iter()
                .filter(|m| m.po.created_at < before)
                .take(limit)
                .collect()
        }
        (None, Some(after)) => {
            // 下拉轮询：created_at > after
            messages
                .into_iter()
                .filter(|m| m.po.created_at > after)
                .collect()
        }
        (Some(before), Some(after)) => {
            // 两者都传：after < created_at < before（理论上不应出现，但兼容）
            messages
                .into_iter()
                .filter(|m| m.po.created_at > after && m.po.created_at < before)
                .collect()
        }
        (None, None) => {
            // 初始加载：返回最新的 limit 条
            messages.into_iter().take(limit).collect()
        }
    };

    // DESC 模式下反转，保证前端显示顺序为 ASC（从旧到新）
    let mut sorted = filtered;
    if params.after_timestamp.is_none() {
        sorted.reverse();
    }

    let total = sorted.len();
    let messages: Vec<MessageListItem> = sorted
        .iter()
        .map(|m| MessageListItem {
            message_id: m.po.id.clone(),
            project_id: m.po.project_id.clone(),
            task_id: m.po.task_id.clone(),
            from_id: m.po.from_id.clone(),
            from_role: m.po.from_role as i32,
            to_id: m.po.to_id.clone(),
            to_role: m.po.to_role as i32,
            message_type: m.po.message_type as i32,
            status: m.po.status as i32,
            content: m.po.content.clone(),
            reply_to_id: m.po.reply_to_id.clone(),
            created_at: m.po.created_at,
        })
        .collect();

    Ok(ListMessagesResponse { messages, total })
}
```

```rust
pub async fn send_message_to_agent(
    ctx: RequestContext,
    params: SendMessageToAgentParams,
) -> Result<SendMessageToAgentResponse> {
    // 判断发送者身份：优先 Agent，其次 User，最后 System
    let (from_id, from_role) = if let Some(aid) = ctx.agent_id() {
        (aid.to_string(), MessageRole::Agent)
    } else if !ctx.uid().is_empty() {
        (ctx.uid(), MessageRole::User)
    } else {
        ("system".to_string(), MessageRole::System)
    };

    let cmd = SendToAgentCommand {
        from_id: &from_id,
        from_role,
        to_agent_id: &params.to_agent_id,
        content: &params.content,
        project_id: params.project_id.as_deref(),
        task_id: params.task_id.as_deref(),
        reply_to_id: params.reply_to_id.as_deref(),
    };

    let message = message::domain().delivery().send_to_agent(ctx, cmd).await?;

    Ok(SendMessageToAgentResponse {
        message_id: message.po.id,
    })
}
```

```rust
/// 加载最新消息（初始加载，默认 10 条）
pub async fn load_latest_messages(
    project_id: Option<&str>,
    limit: Option<usize>,
) -> Result<ListMessagesResponse, String> {
    let mut params = Vec::new();
    if let Some(pid) = project_id {
        params.push(format!("project_id={}", url_encode(pid)));
    }
    if let Some(l) = limit {
        params.push(format!("limit={}", l));
    }
    let query = if params.is_empty() { String::new() } else { format!("?{}", params.join("&")) };
    api_get(&format!("/api/v1/messages{}", query)).await
}

/// 上拉翻页：加载更早的消息
pub async fn load_older_messages(
    project_id: Option<&str>,
    before_timestamp: i64,
    limit: Option<usize>,
) -> Result<ListMessagesResponse, String> {
    let mut params = vec![format!("before_timestamp={}", before_timestamp)];
    if let Some(pid) = project_id {
        params.push(format!("project_id={}", url_encode(pid)));
    }
    if let Some(l) = limit {
        params.push(format!("limit={}", l));
    }
    api_get(&format!("/api/v1/messages?{}", params.join("&"))).await
}

/// 下拉轮询：增量拉取新消息
pub async fn poll_new_messages(
    project_id: Option<&str>,
    after_timestamp: i64,
) -> Result<ListMessagesResponse, String> {
    let mut params = vec![format!("after_timestamp={}", after_timestamp)];
    if let Some(pid) = project_id {
        params.push(format!("project_id={}", url_encode(pid)));
    }
    api_get(&format!("/api/v1/messages?{}", params.join("&"))).await
}

/// URL 编码辅助函数
fn url_encode(s: &str) -> String {
    s.replace('%', "%25")
        .replace(' ', "%20")
        .replace('&', "%26")
        .replace('=', "%3D")
        .replace('+', "%2B")
        .replace('#', "%23")
        .replace('?', "%3F")
}
```

```rust
use std::time::{Duration, SystemTime, UNIX_EPOCH};

use crate::store::auth::use_auth_state;

/// 当前时间戳（毫秒）
fn now_ms() -> i64 {
    SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .unwrap_or_default()
        .as_millis() as i64
}

/// 格式化时间戳为 HH:MM
fn format_time(ts: i64) -> String {
    let secs = ts / 1000;
    let h = (secs / 3600) % 24;
    let m = (secs / 60) % 60;
    format!("{:02}:{:02}", h, m)
}

/// 消息角色（对应后端 MessageRole: 0=User, 1=Agent, 2=System）
fn role_label(role: i32) -> &'static str {
    match role {
        0 => "用户",
        1 => "Agent",
        _ => "系统",
    }
}

fn role_avatar(role: i32) -> &'static str {
    match role {
        0 => "user",
        1 => "agent",
        _ => "system",
    }
}

fn role_initial(role: i32, username: &str) -> String {
    match role {
        0 => username.chars().next().unwrap_or('U').to_string(),
        1 => "A".to_string(),
        _ => "S".to_string(),
    }
}

/// 对话页主组件
#[component]
pub fn Chat() -> Element {
    let auth = use_auth_state();

    // 如果未登录，渲染 Reception 登录页
    if !auth().is_logged_in() {
        return rsx! { crate::pages::reception::Reception() };
    }

    // 项目列表
    let mut projects = use_signal(Vec::new);
    let mut selected_project_id = use_signal(|| None::<String>);
    let mut loading_projects = use_signal(|| true);

    // 消息列表（始终按 created_at ASC 排序：旧 → 新）
    let mut messages = use_signal(Vec::<MessageListItem>::new);
    let mut loading_messages = use_signal(|| true);
    let mut loading_older = use_signal(|| false);
    let mut has_more_older = use_signal(|| true); // 是否还有更早的消息

    // 输入框
    let mut input_text = use_signal(String::new);
    let mut sending = use_signal(|| false);
    let mut error = use_signal(String::new);

    // 当前对话的 Agent ID（MVP: 使用一个默认值或从项目获取）
    let mut current_agent_id = use_signal(|| String::new());

    // 辅助：获取当前消息列表中最老和最新的时间戳
    let oldest_timestamp = || {
        messages().first().map(|m| m.created_at).unwrap_or(0)
    };
    let newest_timestamp = || {
        messages().last().map(|m| m.created_at).unwrap_or(0)
    };

    // 加载项目列表
    let load_projects = move || {
        spawn(async move {
            match list_projects().await {
                Ok(resp: ListProjectsResponse) => {
                    projects.set(resp.projects);
                }
                Err(e) => {
                    error.set(format!("加载项目列表失败: {}", e));
                }
            }
            loading_projects.set(false);
        });
    };

    // 初始加载：获取最新 10 条消息
    let load_latest = move |project_id: Option<String>| {
        spawn(async move {
            let pid_ref = project_id.as_deref();
            match load_latest_messages(pid_ref, Some(10)).await {
                Ok(resp) => {
                    messages.set(resp.messages);
                    has_more_older.set(resp.total >= 10);
                    loading_messages.set(false);
                }
                Err(e) => {
                    error.set(format!("加载消息失败: {}", e));
                    loading_messages.set(false);
                }
            }
        });
    };

    // 上拉翻页：加载更早的消息
    let load_older = move |project_id: Option<String>| {
        if loading_older() || !has_more_older() {
            return;
        }
        let oldest = oldest_timestamp();
        if oldest == 0 {
            return;
        }
        loading_older.set(true);
        spawn(async move {
            let pid_ref = project_id.as_deref();
            match load_older_messages(pid_ref, oldest, Some(10)).await {
                Ok(resp) => {
                    if resp.messages.is_empty() {
                        has_more_older.set(false);
                    } else {
                        // 新加载的旧消息前置到列表
                        let mut current = messages();
                        let mut older = resp.messages;
                        older.extend(current);
                        messages.set(older);
                    }
                    loading_older.set(false);
                }
                Err(_) => {
                    loading_older.set(false);
                }
            }
        });
    };

    // 下拉轮询：增量拉取新消息
    let poll_new = move |project_id: Option<String>| {
        let newest = newest_timestamp();
        if newest == 0 {
            return;
        }
        spawn(async move {
            let pid_ref = project_id.as_deref();
            match poll_new_messages(pid_ref, newest).await {
                Ok(resp) => {
                    if !resp.messages.is_empty() {
                        let mut current = messages();
                        current.extend(resp.messages);
                        messages.set(current);
                    }
                }
                Err(_) => {}
            }
        });
    };

    // 初始加载
    use_effect(move || {
        load_projects();
        load_latest(None);
    });

    // 轮询：每 3 秒拉取新消息
    use_future(move || async move {
        loop {
            tokio::time::sleep(Duration::from_secs(3)).await;
            let pid = selected_project_id();
            poll_new(pid);
        }
    });

    // 选择项目
    let on_select_project = move |pid: String| {
        selected_project_id.set(Some(pid.clone()));
        loading_messages.set(true);
        error.set(String::new());
        has_more_older.set(true);
        load_latest(Some(pid));
    };

    // 返回前台对话
    let on_back_to_general = move |_| {
        selected_project_id.set(None);
        loading_messages.set(true);
        error.set(String::new());
        has_more_older.set(true);
        load_latest(None);
    };

    // 上拉滚动事件
    let on_scroll = move |e: Event<dom::HtmlDivElement>| {
        let el = e.data().downcast::<web_sys::HtmlDivElement>().unwrap();
        // 滚动到顶部时加载更早消息
        if el.scroll_top() < 50 {
            let pid = selected_project_id();
            load_older(pid);
        }
    };

    // 发送消息
    let on_send = move |_| {
        let text = input_text();
        if text.trim().is_empty() || sending() {
            return;
        }
        if current_agent_id().is_empty() {
            error.set("请先选择对话的 Agent".to_string());
            return;
        }

        let agent_id = current_agent_id();
        let project_id = selected_project_id();
        let content = text.clone();

        spawn(async move {
            sending.set(true);
            error.set(String::new());

            let req = SendMessageToAgentParams {
                to_agent_id: agent_id,
                content,
                project_id,
                task_id: None,
                reply_to_id: None,
            };

            match send_message_to_agent(req).await {
                Ok(_resp) => {
                    input_text.set(String::new());
                    // 立即拉取一次新消息
                    let pid = selected_project_id();
                    poll_new(pid);
                }
                Err(e) => {
                    error.set(format!("发送失败: {}", e));
                }
            }
            sending.set(false);
        });
    };

    // 键盘 Enter 发送（Shift+Enter 换行）
    let on_keydown = move |e: KeyboardEvent| {
        if e.key() == Key::Enter && !e.modifiers().shift() {
            e.prevent_default();
            on_send(());
        }
    };

    rsx! {
        div { class: "chat-layout",
            // ===== 左侧：项目列表 =====
            div { class: "chat-sidebar",
                div { class: "chat-sidebar-header",
                    span { "对话" }
                    button {
                        class: "btn btn-ghost btn-sm",
                        onclick: on_back_to_general,
                        "+ 新对话"
                    }
                }
                div { class: "chat-sidebar-list",
                    if loading_projects() {
                        div { class: "chat-loading", "加载项目..." }
                    } else {
                        // 前台对话（无项目）
                        div {
                            class: if selected_project_id().is_none() { "chat-sidebar-item active" } else { "chat-sidebar-item" },
                            onclick: on_back_to_general,
                            "💬 前台对话"
                        }
                        // 项目列表
                        for p in projects() {
                            {
                                let pid = p.id.clone();
                                let is_active = selected_project_id() == Some(pid.clone());
                                let pname = p.name.clone();
                                let item_class = if is_active { "chat-sidebar-item active" } else { "chat-sidebar-item" };
                                rsx! {
                                    div {
                                        key: "{pid}",
                                        class: "{item_class}",
                                        onclick: move |_| on_select_project(pid.clone()),
                                        "📁 {pname}"
                                    }
                                }
                            }
                        }
                    }
                }
            }

            // ===== 右侧：对话区 =====
            div { class: "chat-main",
                // 对话头
                div { class: "chat-header",
                    div {
                        div { class: "chat-header-title",
                            if let Some(_pid) = selected_project_id() {
                                "项目对话"
                            } else {
                                "前台对话"
                            }
                        }
                        div { class: "chat-header-subtitle",
                            if selected_project_id().is_some() {
                                "与 PMO Agent 对话"
                            } else {
                                "与前台 Agent 对话"
                            }
                        }
                    }
                }

                // 错误提示
                if !error().is_empty() {
                    div { style: "padding: var(--space-2) var(--space-4);",
                        ErrorAlert { message: error() }
                    }
                }

                // 消息列表（可滚动，监听上拉事件）
                div { class: "chat-messages", onscroll: on_scroll,
                    if loading_messages() {
                        Loading {}
                    } else if messages().is_empty() {
                        div { class: "chat-loading",
                            "开始对话吧！在下方输入消息，按 Enter 发送"
                        }
                    } else {

                        if loading_older() {
                            div { class: "chat-loading", "加载更早消息..." }
                        } else if !has_more_older() {
                            div { class: "chat-loading", style: "font-size: 12px; opacity: 0.6;", "没有更早的消息了" }

                        // 消息列表
                        for msg in messages() {
                            {
                                let role = msg.from_role;
                                let avatar_class = format!("message-avatar {}", role_avatar(role));
                                let row_class = format!("message-row {}", role_avatar(role));

                                let time_str = format_time(msg.created_at);
                                let role_str = role_label(role);

                                rsx! {
                                    div {
                                        key: "{msg.message_id}",
                                        class: "{row_class}",
                                        div {
                                            class: "{avatar_class}",
                                            "{avatar_text}"
                                        }

                                            div { class: "{bubble_class}", "{content}" }

                div { class: "chat-input-area",

                }
            }
        }
    }
```

## 设计决策记录

### 1. 后端分层遵循

- **Handler** → 调用 `message::domain().management().query()` 和 `message::domain().delivery().send_to_agent()`
- **Domain** → `MessageManagement::query()` 透传 DAL（已有，无需修改）
- **DAL** → `MessageDal::query()` 透传 DAO（已有，无需修改）
- **DAO** → `MessageDao::query()` 执行 SQL（已有，仅新增 organization_id 过滤）

**没有跨层调用**：Handler 不直接调用 DAL/DAO，Domain 不直接调用 DAO。

### 2. 路由设计

- `/` → Chat（对话首页，未登录时内嵌 Reception 登录组件）
- `/login` → Reception（独立的登录/初始化页）
- `/messages/chat` 路由已删除（对话功能合并到首页）

### 3. 双向分页机制

**初始加载**：`GET /api/v1/messages?limit=10` → 默认 DESC 排序取最新 10 条，Handler 层反转为 ASC 返回
**上拉翻页**：`GET /api/v1/messages?before_timestamp={oldest}&limit=10` → 加载更早的消息，前置到列表
**下拉轮询**：`GET /api/v1/messages?after_timestamp={newest}` → 每 3 秒增量拉取新消息，追加到列表末尾
**滚动检测**：`onscroll` 事件监听 `scrollTop < 50` 时触发上拉翻页
**状态追踪**：`has_more_older` signal 标记是否还有更早消息，避免无效请求

### 4. 消息展示

- User 消息：右对齐，橙色气泡
- Agent 消息：左对齐，暖白气泡
- System 消息：左对齐，灰色斜体气泡
- 每条消息显示发送者 + 时间

### 5. Agent 选择

MVP 阶段暂不实现 Agent 自动选择。用户需要在 `current_agent_id` signal 中手动设置目标 Agent ID（后续可从项目信息中获取 PMO Agent ID，或从 Agent 列表中选择）。

### 6. 已知限制

- 无 SSE 实时推送（使用 3 秒短轮询）
- 无消息分页加载（仅加载最近 50 条）→ 已改为双向分页（默认 10 条 + 上拉翻页）
- 无 Markdown 渲染（纯文本展示）
- 无文件/图片消息预览
- Agent 选择需手动设置（后续从项目信息获取）
- `tokio::time::sleep` 在 WASM 中可能不可用 — 如果编译报错，替换为 `gloo_timers::future::TimeoutFuture::new(3000).await`（需在 `frontend/Cargo.toml` 添加 `gloo-timers = { version = "0.3", features = ["futures"] }` 依赖）
